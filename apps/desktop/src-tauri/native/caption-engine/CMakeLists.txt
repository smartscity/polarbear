cmake_minimum_required(VERSION 3.20)
project(polarbear_caption LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "WHISPER_SOURCE must point to the verified pinned source archive")
endif()
add_subdirectory(${WHISPER_SOURCE} whisper-build)
if(APPLE AND TARGET ggml-blas)
  # The helper supports macOS 13.0; the new Accelerate ABI requires macOS 13.3.
  target_compile_options(ggml-blas PRIVATE -UACCELERATE_NEW_LAPACK)
endif()
add_executable(polarbear-caption-engine main.cpp)
target_include_directories(polarbear-caption-engine PRIVATE ${WHISPER_SOURCE}/examples)
target_link_libraries(polarbear-caption-engine PRIVATE whisper)
enable_testing()
find_package(Threads REQUIRED)
target_link_libraries(polarbear-caption-engine PRIVATE Threads::Threads)
# Candidate-only target. Default builds and packaged assets remain unchanged
# until paired accuracy and native latency qualification pass on both CPUs.
if(SHERPA_ONNX_INCLUDE_DIR)
  add_executable(caption-streaming-recognizer-test streaming_recognizer.cpp streaming_recognizer_test.cpp)
  target_include_directories(caption-streaming-recognizer-test PRIVATE ${SHERPA_ONNX_INCLUDE_DIR})
  add_test(NAME caption-streaming-recognizer COMMAND caption-streaming-recognizer-test)
  if(SHERPA_ONNX_C_API_LIBRARY)
    add_executable(polarbear-caption-streaming-candidate main.cpp streaming_runtime.cpp streaming_recognizer.cpp)
    target_compile_definitions(polarbear-caption-streaming-candidate PRIVATE POLARBEAR_STREAMING_CANDIDATE=1)
    target_include_directories(polarbear-caption-streaming-candidate PRIVATE
      ${SHERPA_ONNX_INCLUDE_DIR} ${WHISPER_SOURCE}/examples)
    target_link_libraries(polarbear-caption-streaming-candidate PRIVATE
      whisper Threads::Threads ${SHERPA_ONNX_C_API_LIBRARY})
  endif()
endif()
add_executable(caption-queue-test recognition_queue_test.cpp)
target_link_libraries(caption-queue-test PRIVATE Threads::Threads)
add_test(NAME caption-queue COMMAND caption-queue-test)
add_executable(caption-streaming-input-test streaming_input_test.cpp)
target_link_libraries(caption-streaming-input-test PRIVATE Threads::Threads)
add_test(NAME caption-streaming-input COMMAND caption-streaming-input-test)
set_tests_properties(caption-streaming-input PROPERTIES TIMEOUT 10)
add_executable(caption-streaming-decoder-test streaming_decoder_test.cpp)
target_link_libraries(caption-streaming-decoder-test PRIVATE Threads::Threads)
add_test(NAME caption-streaming-decoder COMMAND caption-streaming-decoder-test)
set_tests_properties(caption-streaming-decoder PROPERTIES TIMEOUT 10)
add_executable(caption-pcm-frames-test pcm_frames_test.cpp)
add_test(NAME caption-pcm-frames COMMAND caption-pcm-frames-test)
add_executable(caption-config-test inference_config_test.cpp)
add_test(NAME caption-config COMMAND caption-config-test)
add_executable(caption-preview-test preview_schedule_test.cpp)
add_test(NAME caption-preview COMMAND caption-preview-test)
add_executable(caption-cancellation-test cancellation_test.cpp)
target_link_libraries(caption-cancellation-test PRIVATE whisper)
add_test(NAME caption-cancellation COMMAND caption-cancellation-test
  "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/caption/ggml-base.en-q5_1.bin")
set_tests_properties(caption-cancellation PROPERTIES TIMEOUT 60)
set_tests_properties(caption-queue PROPERTIES TIMEOUT 10)
install(TARGETS polarbear-caption-engine RUNTIME DESTINATION .)
